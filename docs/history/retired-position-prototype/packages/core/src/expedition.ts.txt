import {copy,requireRule,RuleError} from './battle';
import {deck,evaluate,type Card} from './cards';
import {shuffle,randomState,hash,type RandomState} from './random';
import type {Suit} from '@three-card/content';
export const EXP_HEROES={zhangfei:{name:'张飞',hp:12,text:'先锋或中军：击退最近敌人；堵住退路时碰撞。'},zhaoyun:{name:'赵云',hp:10,text:'先锋空缺时攻击最后方敌人，并与我方先锋换位。'},liubei:{name:'刘备',hp:11,text:'攻击最近敌人，并为相邻友军各恢复2兵力。'},zhugeliang:{name:'诸葛亮',hp:9,text:'让相邻且已行动友军重新待命；同时给他2护甲。'},huangzhong:{name:'黄忠',hp:9,text:'直接射击最后方敌人；在后援时额外2伤害。'},guanyu:{name:'关羽',hp:12,text:'攻击最近敌人；本轮为对子或三同点时再攻击一次。'}} as const;
export type ExpHero=keyof typeof EXP_HEROES;
export const EXP_RELICS={pursuit:{name:'破阵追骑',text:'敌人被击退后，我方先锋追击最近敌人3伤害。让中军张飞开路。'},ram:{name:'震军鼓',text:'碰撞时两名敌人各额外受到3伤害。主动将敌人挤到一起。'},relay:{name:'接力令旗',text:'战中换位后，中军攻击最近敌人3伤害。每回合最多触发2次。'},banner:{name:'同袍甲',text:'分配的三张牌至少两张同兵种时，全军本回合获得2护甲。'},supply:{name:'行军青囊',text:'每回合换牌时，最受伤的武将恢复3兵力。'},echo:{name:'奇谋锦囊',text:'每回合第一次使用谋牌行动后，该武将重新待命。再次出手仍花军令。'}} as const;
export type ExpRelic=keyof typeof EXP_RELICS;
export const EXP_STAGES=[{name:'水寨前哨',text:'盾兵在前，弓手在后；推开盾线可直取弓手。',hp:[10,0,8]},{name:'桥头密阵',text:'敌人挤在一起，击退会制造碰撞。',hp:[13,8,10]},{name:'渡口伏兵',text:'两名弓手夹击后援，及时换位保护伤员。',hp:[12,10,12]},{name:'渠帅亲卫',text:'最后一战，重甲与后排齐射同时施压。',hp:[17,12,14]}] as const;
export interface ExpUnit{id:ExpHero;hp:number;guard:number;acted:boolean;card:string|null}
export interface ExpEnemy{id:'shield'|'archer';hp:number}
export interface ExpState{version:'expedition-1';seq:number;seed:string;rng:RandomState;phase:'starter'|'assign'|'player'|'reward'|'forge'|'victory'|'defeat';stage:number;round:number;commands:number;team:ExpUnit[];enemies:(ExpEnemy|null)[];army:(Card&{boost:number})[];hand:string[];relics:ExpRelic[];offers:string[];category:number;redraw:boolean;relayCount:number;echoUsed:boolean;schemeUsed:string[];freeSwap:boolean;log:string[];nextId:number;stats:{kills:number;triggers:number;edits:number}}
export interface ExpAction{seq:number;type:'starter'|'assign'|'swap'|'redraw'|'commit'|'act'|'end'|'reward'|'forge';id?:string;actor?:ExpHero;target?:ExpHero;card?:string;from?:number;to?:number;edit?:'remove'|'copy'|'boost'|'suit';suit?:Suit;replace?:ExpHero}
const relicIds=Object.keys(EXP_RELICS) as ExpRelic[],heroIds=Object.keys(EXP_HEROES) as ExpHero[];
const unit=(id:ExpHero):ExpUnit=>({id,hp:EXP_HEROES[id].hp,guard:0,acted:false,card:null});
export function newExpedition(seed:string):ExpState{return{version:'expedition-1',seq:0,seed,rng:randomState(seed),phase:'starter',stage:0,round:1,commands:4,team:[unit('zhaoyun'),unit('zhangfei'),unit('liubei')],enemies:[],army:deck().map(c=>({...c,boost:0})),hand:[],relics:[],offers:['pursuit','ram','relay'],category:0,redraw:false,relayCount:0,echoUsed:false,schemeUsed:[],freeSwap:false,log:['先选军宝，决定这一局怎样破阵。'],nextId:0,stats:{kills:0,triggers:0,edits:0}};}
function log(s:ExpState,t:string){s.log.push(t);s.log=s.log.slice(-45);}
function deal(s:ExpState){s.hand=shuffle(s.army.map(c=>c.id),s.rng,'player').slice(0,6);s.team.forEach(u=>{u.acted=false;u.card=null;u.guard=0;});s.commands=4;s.redraw=false;s.echoUsed=false;s.schemeUsed=[];s.relayCount=0;s.freeSwap=false;s.category=0;s.phase='assign';}
function battle(s:ExpState){s.round=1;s.enemies=EXP_STAGES[s.stage].hp.map((hp,i)=>hp?{id:i===0?'shield':'archer',hp}:null);deal(s);log(s,`第${s.stage+1}战 · ${EXP_STAGES[s.stage].name}`);}
function hit(s:ExpState,i:number,damage:number,source:string){const e=s.enemies[i];if(!e)return;e.hp=Math.max(0,e.hp-damage);log(s,`${source} → ${e.id==='shield'?'盾兵':'弓手'} −${damage}${e.hp?'':'，击破'}`);if(!e.hp){s.enemies[i]=null;s.stats.kills++;}}
const first=(s:ExpState)=>s.enemies.findIndex(e=>e!==null);
const last=(s:ExpState)=>s.enemies.reduce((x,e,i)=>e?i:x,-1);
function relay(s:ExpState){if(s.relics.includes('relay')&&s.relayCount<2&&first(s)>=0){s.relayCount++;s.stats.triggers++;hit(s,first(s),3,`接力令旗 · ${EXP_HEROES[s.team[1].id].name}接力`);}}
function check(s:ExpState){if(s.enemies.every(e=>!e)){if(s.stage===3){s.phase='victory';log(s,'四战告捷！你的构筑完成了破阵。');return;}s.team.forEach(u=>u.hp=Math.min(EXP_HEROES[u.id].hp,u.hp+3));const r=shuffle(relicIds.filter(id=>!s.relics.includes(id)),s.rng,'reward').slice(0,2);const h=shuffle(heroIds.filter(id=>!s.team.some(u=>u.id===id)),s.rng,'reward').slice(0,1);s.offers=[...r.map(id=>'relic:'+id),...h.map(id=>'hero:'+id)];s.phase='reward';log(s,'战后各恢复3兵力。选择新军宝或招募武将，再永久整编。');}}
function apply(s:ExpState,a:ExpAction){requireRule(a.seq===s.seq,'行动已处理');switch(a.type){
case 'starter':requireRule(s.phase==='starter'&&a.id&&s.offers.includes(a.id),'请选择开局军宝');s.relics.push(a.id as ExpRelic);s.offers=[];battle(s);return;
case 'assign':{requireRule(s.phase==='assign','本轮已经出阵');const u=s.team.find(u=>u.id===a.actor);requireRule(u&&a.card&&s.hand.includes(a.card),'请选择武将和手牌');for(const other of s.team)if(other.card===a.card)other.card=null;u.card=a.card;return;}
case 'swap':{requireRule(s.phase==='assign'||s.phase==='player','此时不能换位');requireRule(Number.isInteger(a.from)&&Number.isInteger(a.to)&&a.from!>=0&&a.from!<3&&a.to!>=0&&a.to!<3&&a.from!==a.to,'位置无效');if(s.phase==='player'){requireRule(s.commands>0||s.freeSwap,'军令不足');if(s.freeSwap)s.freeSwap=false;else s.commands--;}[s.team[a.from!],s.team[a.to!]]=[s.team[a.to!],s.team[a.from!]];log(s,'调整阵位，阵牌与行动状态跟随武将。');if(s.phase==='player'){relay(s);check(s);}return;}
case 'redraw':{requireRule(s.phase==='assign'&&!s.redraw,'本回合只能换牌一次');const reserved=s.team.map(u=>u.card).filter((id):id is string=>!!id),remaining=s.army.filter(c=>!s.hand.includes(c.id)).map(c=>c.id);s.hand=[...reserved,...shuffle(remaining,s.rng,'player').slice(0,6-reserved.length)];s.redraw=true;if(s.relics.includes('supply')){const u=[...s.team].sort((a,b)=>(EXP_HEROES[b.id].hp-b.hp)-(EXP_HEROES[a.id].hp-a.hp))[0];u.hp=Math.min(EXP_HEROES[u.id].hp,u.hp+3);log(s,`青囊换牌 · ${EXP_HEROES[u.id].name}恢复3。`);}return;}
case 'commit':{requireRule(s.phase==='assign'&&s.team.every(u=>u.card),'请为三名武将各分配一张牌');s.category=evaluate(s.team.map(u=>s.army.find(c=>c.id===u.card)!)).category;s.freeSwap=[2,4].includes(s.category);const suits=new Set(s.team.map(u=>s.army.find(c=>c.id===u.card)!.suit));const guard=([3,4].includes(s.category)?2:0)+(s.relics.includes('banner')&&suits.size<=2?2:0);s.team.forEach(u=>u.guard=guard);s.phase='player';log(s,`三军出阵：${[1,5].includes(s.category)?'同点配合，攻击增强':s.freeSwap?'连阵，首次主动换位免费':guard?'同袍护甲生效':'按各自阵牌行动'}。`);return;}
case 'act':{requireRule(s.phase==='player'&&s.commands>0,'请先分配阵牌或检查军令');const i=s.team.findIndex(u=>u.id===a.actor);requireRule(i>=0&&!s.team[i].acted,'该武将已经行动');const u=s.team[i],c=s.army.find(c=>c.id===u.card)!;const damage=2+Math.floor(c.rank/3)+c.boost+(c.suit==='spear'?1:0)+(s.category===5?2:s.category===1?1:0);const attack=(target:number,amount:number,source:string)=>hit(s,target,amount+(c.suit==='bow'&&target===last(s)?2:0),source);
if(u.id==='zhangfei'){requireRule(i<2,'张飞在先锋或中军才能喝退');const n=first(s);attack(n,damage,'张飞喝退');if(s.enemies[n]){if(n===2)hit(s,n,2,'退无可退');else if(s.enemies[n+1]){const extra=s.relics.includes('ram')?3:0;if(extra)s.stats.triggers++;hit(s,n,2+extra,extra?'震军鼓碰撞':'碰撞');hit(s,n+1,2+extra,extra?'震军鼓碰撞':'碰撞');}else{s.enemies[n+1]=s.enemies[n];s.enemies[n]=null;log(s,'盾线后移，先锋通道打开。');if(s.relics.includes('pursuit')){s.stats.triggers++;hit(s,first(s),3,`破阵追骑 · ${EXP_HEROES[s.team[0].id].name}追击`);}}}}
else if(u.id==='zhaoyun'){attack(s.enemies[0]?first(s):last(s),damage+(!s.enemies[0]?2:0),'赵云突进');if(i!==0){[s.team[0],s.team[i]]=[s.team[i],s.team[0]];log(s,'赵云换到先锋，改变后续支援与受击目标。');relay(s);}}
else if(u.id==='zhugeliang'){const t=s.team.findIndex(x=>x.id===a.target);requireRule(t>=0&&Math.abs(t-i)===1&&s.team[t].acted,'选择相邻、已行动友军');s.team[t].acted=false;s.team[t].guard+=2;log(s,`诸葛亮再令 · ${EXP_HEROES[s.team[t].id].name}再次待命并获得2护甲。`);}
else if(u.id==='huangzhong')attack(last(s),damage+(i===2?2:0),'黄忠后排点杀');
else if(u.id==='liubei'){attack(first(s),damage,'刘备合击');s.team.forEach((t,j)=>{if(Math.abs(i-j)===1)t.hp=Math.min(EXP_HEROES[t.id].hp,t.hp+2);});log(s,'刘备援护：相邻友军恢复2兵力。');}
else{attack(first(s),damage,'关羽斩击');if([1,5].includes(s.category)&&first(s)>=0)attack(first(s),damage,'关羽同点连斩');}
u.acted=true;s.commands--;if(c.suit==='cavalry')u.guard+=2;if(c.suit==='scheme'&&!s.schemeUsed.includes(c.id)){s.schemeUsed.push(c.id);s.commands++;log(s,'谋牌调度：返还1军令（每张牌每回合一次）。');}if(s.relics.includes('echo')&&c.suit==='scheme'&&!s.echoUsed){s.echoUsed=true;u.acted=false;s.stats.triggers++;log(s,'奇谋锦囊：本回合首次谋牌行动者重新待命。');}check(s);return;}
case 'end':{requireRule(s.phase==='player','请先出阵');for(const e of s.enemies){if(!e)continue;const u=s.team[e.id==='shield'?0:2],raw=(e.id==='shield'?3:2)+Math.floor(s.stage/2),damage=Math.max(0,raw-u.guard);u.guard=Math.max(0,u.guard-raw);u.hp=Math.max(0,u.hp-damage);log(s,`${e.id==='shield'?'盾兵反击':'弓手齐射'} → ${EXP_HEROES[u.id].name} −${damage}`);if(!u.hp){s.phase='defeat';return;}}if(s.round>=3){s.phase='defeat';log(s,'敌军援军到达，本场三回合耗尽。');return;}s.round++;deal(s);return;}
case 'reward':{requireRule(s.phase==='reward'&&a.id&&s.offers.includes(a.id),'奖励已过期');const[kind,id]=a.id.split(':');if(kind==='relic')s.relics.push(id as ExpRelic);else{const i=s.team.findIndex(u=>u.id===a.replace);requireRule(i>=0,'请选择离队武将');s.team[i]=unit(id as ExpHero);}s.phase='forge';s.offers=[];return;}
case 'forge':{requireRule(s.phase==='forge','当前不在整编');const i=s.army.findIndex(c=>c.id===a.card);requireRule(i>=0,'请选择阵牌');if(a.edit==='remove'){requireRule(s.army.length>12,'至少保留12张');s.army.splice(i,1);}else if(a.edit==='copy')s.army.push({...s.army[i],id:'recruit-'+s.nextId++});else if(a.edit==='boost')s.army[i].boost++;else{requireRule(a.edit==='suit'&&a.suit&&['spear','cavalry','bow','scheme'].includes(a.suit),'请选择兵种');s.army[i].suit=a.suit;}s.stats.edits++;s.stage++;battle(s);return;}
}throw new RuleError('未知行动');}
export function expeditionAction(state:ExpState,a:ExpAction):{state:ExpState;error?:string}{const s=copy(state);try{apply(s,a);s.seq++;return{state:s};}catch(e){if(e instanceof RuleError)return{state,error:e.message};throw e;}}
export function expeditionView(s:ExpState){const{rng,...v}=s;return copy({...v,hand:s.hand.map(id=>s.army.find(c=>c.id===id)!)})}
export function expeditionIntents(s:ExpState){return s.enemies.filter(e=>e!==null).map(e=>{const u=s.team[e!.id==='shield'?0:2],raw=(e!.id==='shield'?3:2)+Math.floor(s.stage/2);return`${e!.id==='shield'?'盾兵':'弓手'} → ${EXP_HEROES[u.id].name}，${raw}伤害（护甲可减免）`;});}
export function expSerialize(s:ExpState){const payload=JSON.stringify(s);return JSON.stringify({payload,checksum:hash(payload)});}
export function expRestore(raw:string|null):ExpState|null{try{const e=JSON.parse(raw??'null');if(!e||hash(e.payload)!==e.checksum)return null;const s=JSON.parse(e.payload);if(s.version!=='expedition-1'||!Number.isInteger(s.seq)||s.stage<0||s.stage>3||!Array.isArray(s.team)||s.team.length!==3||s.team.some((u:ExpUnit)=>!Object.hasOwn(EXP_HEROES,u.id))||!Array.isArray(s.army)||!Array.isArray(s.log)||!s.rng)return null;return s;}catch{return null;}}
