import {copy,requireRule,RuleError} from './battle';
export type GeneralId='zhangfei'|'zhaoyun'|'zhugeliang';
export interface Unit {id:GeneralId;hp:number;acted:boolean}
export interface TacticalState {seq:number;phase:'deploy'|'player'|'victory'|'defeat';round:number;commands:number;allies:Unit[];enemies:({id:'shield'|'archer';hp:number}|null)[];log:string[];history:string[]}
export type TacticalAction={seq:number;type:'swap'|'begin'|'skill'|'attack'|'end';from?:number;to?:number;actor?:GeneralId;target?:GeneralId};
export const GENERALS={zhangfei:{name:'张飞',hp:8,skill:'喝退',text:'仅先锋可用：造成2伤害，将最近敌人击退一格。后格被占则碰撞，双方再受2伤害。'},zhaoyun:{name:'赵云',hp:6,skill:'龙胆突进',text:'敌先锋空缺时突袭最后方敌人4伤害，否则攻击最近敌人2伤害。随后与我方先锋换位。'},zhugeliang:{name:'诸葛亮',hp:5,skill:'再令',text:'让相邻且已经行动的友将再次待命；仍需花军令让他行动。自己不能被再令。'}} as const;
export function newTactics():TacticalState{return{seq:0,phase:'deploy',round:1,commands:4,allies:(['zhangfei','zhugeliang','zhaoyun'] as GeneralId[]).map(id=>({id,hp:GENERALS[id].hp,acted:false})),enemies:[{id:'shield',hp:6},null,{id:'archer',hp:5}],log:['盾兵挡住通道；弓手准备射击我方后援。'],history:[]};}
function hit(s:TacticalState,index:number,damage:number){const enemy=s.enemies[index];if(!enemy)return;enemy.hp=Math.max(0,enemy.hp-damage);s.log.push(`${enemy.id==='shield'?'盾兵':'弓手'}受到${damage}伤害，剩余${enemy.hp}。`);if(!enemy.hp)s.enemies[index]=null;}
function apply(s:TacticalState,a:TacticalAction){requireRule(a.seq===s.seq,'行动已过期');requireRule(s.phase==='deploy'||s.phase==='player','试验已结束，请重新摆阵');
 if(a.type==='swap'){const f=a.from,t=a.to;requireRule(Number.isInteger(f)&&Number.isInteger(t)&&f!>=0&&f!<3&&t!>=0&&t!<3&&f!==t,'请选择两个不同位置');if(s.phase==='player'){requireRule(s.commands>0,'军令不足');s.commands--;}[s.allies[f!],s.allies[t!]]=[s.allies[t!],s.allies[f!]];s.log.push(`换位：${GENERALS[s.allies[f!].id].name}至${['先锋','中军','后援'][f!]}。`);return;}
 if(a.type==='begin'){requireRule(s.phase==='deploy','已经开战');s.phase='player';s.log.push('第1回合开始：4军令，每位武将先行动一次。');return;}
 requireRule(s.phase==='player','请先开始破阵');
 if(a.type==='end'){for(const enemy of s.enemies){if(!enemy)continue;const index=enemy.id==='shield'?0:2,damage=enemy.id==='shield'?(s.enemies[0]?.id==='shield'?3:2):4;const ally=s.allies[index];ally.hp=Math.max(0,ally.hp-damage);s.log.push(`${enemy.id==='shield'?'盾兵反击':'弓手齐射'} → ${GENERALS[ally.id].name}受到${damage}伤害。`);if(!ally.hp){s.phase='defeat';s.log.push('一名武将倒下，破阵失败。');return;}}
  if(s.round>=3){s.phase='defeat';s.log.push('敌军援兵赶到，三回合破阵期限已过。');return;}s.round++;s.commands=4;s.allies.forEach(u=>u.acted=false);s.log.push(`第${s.round}回合：军令与行动次数恢复，站位保留。`);return;}
 requireRule(a.type==='skill'||a.type==='attack','未知行动');const index=s.allies.findIndex(u=>u.id===a.actor);requireRule(index>=0,'请选择武将');const unit=s.allies[index];requireRule(s.commands>0&&!unit.acted,'军令不足或该武将已行动');
 const nearest=s.enemies.findIndex(e=>e!==null);requireRule(nearest>=0,'敌军已被消灭');
 if(a.type==='attack'){hit(s,nearest,2);s.log.push(`${GENERALS[unit.id].name}普通攻击。`);}
 else if(unit.id==='zhangfei'){requireRule(index===0,'张飞需要站在先锋才能喝退');s.log.push('张飞喝退：打开通道，或制造碰撞。');hit(s,nearest,2);if(s.enemies[nearest]){if(nearest===2){hit(s,nearest,2);s.log.push('退无可退：追加2伤害。');}else if(s.enemies[nearest+1]){s.log.push('敌军碰撞：两名敌人各受2伤害。');hit(s,nearest,2);hit(s,nearest+1,2);}else{s.enemies[nearest+1]=s.enemies[nearest];s.enemies[nearest]=null;s.log.push('盾线后移，先锋位置出现缺口。');}}}
 else if(unit.id==='zhaoyun'){const open=!s.enemies[0],target=open?s.enemies.reduce((last,e,i)=>e?i:last,-1):nearest;s.log.push(open?'赵云穿过缺口，突袭最后方敌人！':'通道被阻：赵云正面交战。');hit(s,target,open?4:2);if(index!==0){[s.allies[0],s.allies[index]]=[s.allies[index],s.allies[0]];s.log.push(`赵云突进至先锋，${GENERALS[s.allies[index].id].name}换到${['先锋','中军','后援'][index]}。`);}}
 else{const t=s.allies.findIndex(u=>u.id===a.target);requireRule(t>=0&&Math.abs(t-index)===1&&s.allies[t].acted,'请选择相邻且已行动的友将');requireRule(s.allies[t].id!=='zhugeliang','不能对自己再令');s.allies[t].acted=false;s.log.push(`诸葛亮再令：${GENERALS[s.allies[t].id].name}重新待命。`);}
 unit.acted=true;s.commands--;if(s.enemies.every(e=>!e)){s.phase='victory';s.log.push(`第${s.round}回合破阵成功，剩余${s.commands}军令。`);}
}
export function tacticalAction(state:TacticalState,action:TacticalAction):{state:TacticalState;error?:string}{const s=copy(state);try{apply(s,action);s.seq++;s.log=s.log.slice(-60);return{state:s};}catch(e){if(e instanceof RuleError)return{state,error:e.message};throw e;}}
export function enemyIntent(s:TacticalState){return s.enemies.filter(e=>e!==null).map(e=>`${e!.id==='shield'?'盾兵攻击先锋':'弓手攻击后援'}：${GENERALS[s.allies[e!.id==='shield'?0:2].id].name}将受${e!.id==='shield'?(s.enemies[0]?.id==='shield'?3:2):4}伤害`);}
