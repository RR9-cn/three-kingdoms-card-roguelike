import {_electron as electron} from '@playwright/test';
import {writeFile,mkdtemp} from 'node:fs/promises';
import {tmpdir} from 'node:os';import path from 'node:path';import assert from 'node:assert/strict';
const userData=await mkdtemp(path.join(tmpdir(),'tactical-smoke-'));
const app=await electron.launch({executablePath:path.resolve('release/ThreeCardKingdoms-darwin-arm64/ThreeCardKingdoms.app/Contents/MacOS/ThreeCardKingdoms'),args:[],env:{...process.env,THREE_CARD_USER_DATA:userData}});
try{const p=await app.firstWindow();await p.waitForSelector('[data-action=begin]');const errors=[];p.on('pageerror',e=>errors.push(e.message));const click=a=>p.locator(`[data-action="${a}"]`).click();await click('begin');await click('skill');await click('select:zhugeliang');await click('skill:zhangfei');await click('select:zhangfei');await click('skill');await click('select:zhaoyun');await click('skill');assert.equal(await p.evaluate(()=>JSON.parse(window.render_tactics_to_text()).phase),'victory');await p.screenshot({path:'docs/evidence/tactics-native.png'});await p.locator('a[href="./index.html"]').click();await p.waitForSelector('#seed');assert.deepEqual(errors,[]);await writeFile('docs/evidence/tactics-native.json',JSON.stringify({platform:process.platform,packaged:true,victory:true,returnToBuild:true,errors},null,2));console.log('Native tactical victory and build-mode navigation passed.');}finally{await app.close();}
