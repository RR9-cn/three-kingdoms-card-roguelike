import {test,expect,type Page} from '@playwright/test';
const click=(p:Page,a:string)=>p.locator(`[data-action="${a}"]`).click();
const skill=async(p:Page,id:string,target?:string)=>{await click(p,'select:'+id);await click(p,target?'skill:'+target:'skill');};
const state=(p:Page)=>p.evaluate(()=>JSON.parse((window as any).render_tactics_to_text()));
test('two tactical solutions, undo and viewport',async({page})=>{await page.goto('http://127.0.0.1:4173/battlefield.html');await page.setViewportSize({width:1280,height:720});await page.screenshot({path:'docs/evidence/tactics-deploy.png'});await click(page,'begin');await skill(page,'zhangfei');expect((await state(page)).enemies[0]).toBeNull();await page.screenshot({path:'docs/evidence/tactics-gap.png'});await skill(page,'zhugeliang','zhangfei');await skill(page,'zhangfei');await skill(page,'zhaoyun');expect((await state(page)).phase).toBe('victory');await page.screenshot({path:'docs/evidence/tactics-collision-victory.png'});await click(page,'undo');expect((await state(page)).phase).toBe('player');await click(page,'reset');await click(page,'begin');await skill(page,'zhangfei');await skill(page,'zhaoyun');await skill(page,'zhugeliang','zhaoyun');await skill(page,'zhaoyun');await click(page,'end');expect((await state(page)).allies[0].hp).toBe(4);await skill(page,'zhaoyun');expect((await state(page)).phase).toBe('victory');expect((await state(page)).round).toBe(2);expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth)).toBe(true);});
