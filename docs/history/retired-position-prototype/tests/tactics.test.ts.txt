import {test} from 'node:test';import assert from 'node:assert/strict';
import {newTactics,tacticalAction,enemyIntent,type TacticalState,type TacticalAction} from '@three-card/core';
const act=(s:TacticalState,a:Omit<TacticalAction,'seq'>)=>{const r=tacticalAction(s,{...a,seq:s.seq});assert.equal(r.error,undefined);return r.state;};
test('collision route: double Zhang Fei breaks shield and Zhao Yun finishes in round one',()=>{let s=act(newTactics(),{type:'begin'});for(const a of [{type:'skill',actor:'zhangfei'},{type:'skill',actor:'zhugeliang',target:'zhangfei'},{type:'skill',actor:'zhangfei'},{type:'skill',actor:'zhaoyun'}] as const)s=act(s,a);assert.equal(s.phase,'victory');assert.equal(s.round,1);assert.equal(s.commands,0);assert.ok(s.log.some(l=>l.includes('碰撞')));});
test('flank route kills archer first and survives a different enemy response',()=>{let s=act(newTactics(),{type:'begin'});for(const a of [{type:'skill',actor:'zhangfei'},{type:'skill',actor:'zhaoyun'},{type:'skill',actor:'zhugeliang',target:'zhaoyun'},{type:'skill',actor:'zhaoyun'}] as const)s=act(s,a);assert.equal(s.phase,'player');assert.ok(s.enemies.some(e=>e?.id==='shield'));assert.ok(!s.enemies.some(e=>e?.id==='archer'));assert.equal(s.allies[0].id,'zhaoyun');assert.deepEqual(enemyIntent(s),['盾兵攻击先锋：赵云将受2伤害']);s=act(s,{type:'end'});assert.equal(s.allies[0].hp,4);s=act(s,{type:'skill',actor:'zhaoyun'});assert.equal(s.phase,'victory');assert.equal(s.round,2);});
test('position and adjacency restrict actions without spending resources',()=>{let s=act(newTactics(),{type:'swap',from:0,to:2});s=act(s,{type:'begin'});assert.equal(tacticalAction(s,{seq:s.seq,type:'skill',actor:'zhangfei'}).state,s);assert.equal(tacticalAction(s,{seq:s.seq,type:'skill',actor:'zhugeliang',target:'zhaoyun'}).state,s);});
test('swapping changes announced rear target and keeps acted flag with general',()=>{let s=act(newTactics(),{type:'begin'});s=act(s,{type:'attack',actor:'zhangfei'});s=act(s,{type:'swap',from:0,to:2});assert.equal(s.allies[2].acted,true);assert.ok(enemyIntent(s).includes('弓手攻击后援：张飞将受4伤害'));assert.equal(s.commands,2);assert.equal(tacticalAction(s,{seq:s.seq-1,type:'end'}).state,s);});
test('enemy phase kills exposed rear and makes failure explicit',()=>{let s=act(newTactics(),{type:'begin'});s=act(s,{type:'end'});s=act(s,{type:'end'});assert.equal(s.phase,'defeat');assert.equal(s.allies[2].hp,0);});
